cmake_minimum_required(VERSION 3.20)
project(LionKingPridePlayer C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(_GNU_SOURCE)
get_filename_component(TLK_PLAYER_ENGINE "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
get_filename_component(TLK_PLAYER_WORKSPACE "${TLK_PLAYER_ENGINE}/.." ABSOLUTE)
set(TLK_PRIDE_ROM "" CACHE FILEPATH "Verified SNES USA ROM used only for local generation")
set(TLK_PRIDE_GENERATION "${CMAKE_CURRENT_BINARY_DIR}/game-program" CACHE PATH "Private generated game programs")
set(TLK_PRIDE_MD_BACKEND "${TLK_PLAYER_WORKSPACE}/.tools/genesis-plus-gx/genesis_plus_gx_libretro.dll" CACHE FILEPATH "Reviewed MD runtime backend")
if(NOT EXISTS "${TLK_PRIDE_ROM}")
    message(FATAL_ERROR "Set TLK_PRIDE_ROM to the verified local SNES USA ROM")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT EXISTS "${TLK_PRIDE_GENERATION}/native/program_manifest.json")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${TLK_PLAYER_ENGINE}/scripts/generate_pride_player.py"
        --rom "${TLK_PRIDE_ROM}" --out "${TLK_PRIDE_GENERATION}" RESULT_VARIABLE result)
    if(NOT result EQUAL 0)
        message(FATAL_ERROR "Standalone player program generation failed: ${result}")
    endif()
endif()
set(TLK_SNESRECOMP_GEN_DIR "${TLK_PRIDE_GENERATION}/native" CACHE PATH "Native baseline program" FORCE)
set(CMAKE_PROJECT_TLKRecomp_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/deferred-player.cmake")
add_subdirectory("${TLK_PLAYER_ENGINE}" native-engine EXCLUDE_FROM_ALL)
